# Додаток 1. Концептуальна схема бази даних (Нотація Пітера Чена)

У цій концептуальній схемі (ER-діаграмі) у нотації Пітера Чена сутності зображені прямокутниками, їхні зв'язки — ромбами, а атрибути — овалами (з підкресленими первинними ключами `PK`). Зв'язок багато-до-багатьох (M:N) між Користувачем та Досягненням має власний атрибут зв'язку `earned_at`.

```mermaid
flowchart TD
    %% Стилізація
    classDef entity fill:#d4f7d4,stroke:#333,stroke-width:2px;
    classDef attribute fill:#fff0f0,stroke:#333,stroke-width:1px,rx:20,ry:20;
    classDef relationship fill:#e6f2ff,stroke:#333,stroke-width:2px,shape:diamond;

    %% Сутності
    U[Користувач]:::entity
    S[Тема правопису]:::entity
    G[Тренажер]:::entity
    Q[Завдання]:::entity
    A[Досягнення]:::entity
    M[Робота над помилками]:::entity
    R[Результат гри]:::entity

    %% Атрибути Користувача
    U_id([<u>id PK</u>]):::attribute --- U
    U_n([username]):::attribute --- U
    U_p([password]):::attribute --- U
    U_r([role]):::attribute --- U
    U_c([created_at]):::attribute --- U

    %% Атрибути Теми правопису
    S_id([<u>id PK</u>]):::attribute --- S
    S_n([name]):::attribute --- S
    S_d([description]):::attribute --- S

    %% Атрибути Тренажера
    G_id([<u>id PK</u>]):::attribute --- G
    G_n([name]):::attribute --- G
    G_d([description]):::attribute --- G

    %% Атрибути Завдання
    Q_id([<u>id PK</u>]):::attribute --- Q
    Q_t([question_text]):::attribute --- Q
    Q_c([correct_answer]):::attribute --- Q
    Q_s([sort_order]):::attribute --- Q

    %% Атрибути Досягнення
    A_id([<u>id PK</u>]):::attribute --- A
    A_n([name]):::attribute --- A
    A_d([description]):::attribute --- A
    A_b([badge_path]):::attribute --- A
    A_s([system_key]):::attribute --- A

    %% Атрибути Роботи над помилками
    M_id([<u>id PK</u>]):::attribute --- M
    M_t([question_text]):::attribute --- M
    M_c([correct_answer]):::attribute --- M

    %% Атрибути Результату гри
    R_id([<u>id PK</u>]):::attribute --- R
    R_s([score]):::attribute --- R
    R_c([correct_answers]):::attribute --- R
    R_t([total_questions]):::attribute --- R
    R_p([played_at]):::attribute --- R

    %% Зв'язки
    S_G{Містить}:::relationship
    S ---|1| S_G
    S_G ---|N| G

    G_Q{Має}:::relationship
    G ---|1| G_Q
    G_Q ---|N| Q

    U_M{Робить помилку}:::relationship
    U ---|1| U_M
    U_M ---|N| M

    U_R{Має спробу}:::relationship
    U ---|1| U_R
    U_R ---|N| R

    G_R{Записує результат}:::relationship
    G ---|1| G_R
    G_R ---|N| R

    U_A{Отримує досягнення}:::relationship
    U ---|M| U_A
    U_A ---|N| A
    
    %% Атрибут зв'язку "Отримує досягнення"
    Attr_earned([earned_at]):::attribute --- U_A
```
